C++ STL stack — 왜 써야 할까?

세영코딩연구소 · 정올 준비 | C 배열 스택 vs C++ STL stack 완전 비교

STL(Standard Template Library)이란?
C++ 표준 라이브러리에 포함된 검증된 자료구조와 알고리즘의 모음이야.
직접 구현보다 버그 없고, 빠르고, 코드가 짧아서 정올 실전에서 강력해!

C 배열 스택 (직접 구현)

c_stack.c
int stk[100001];
int top = -1;

stk[++top] = 5;   // push
stk[++top] = 3;
int v = stk[top--]; // pop
int t = stk[top];   // peek
int sz = top + 1;   // size
int empty = (top == -1); // empty
❌ 범위 초과 직접 관리 / 코드 길어짐 / 실수 위험

C++ STL stack (권장)

stl_stack.cpp
#include <stack>
using namespace std;
stack<int> stk;

stk.push(5);    // push
stk.push(3);
stk.pop();      // 제거 (반환X)
int t = stk.top(); // peek
int sz = stk.size(); // size
bool e = stk.empty(); // empty
✅ 자동 크기 관리 / 깔끔한 API / 버그 없음

STL stack의 내부 구조

stack<T, Container> — 기본 컨테이너는 deque<T>야.
성능이 중요하면 stack<int, vector<int>>로 vector 기반으로 바꿀 수 있어. 일반적으로 더 빠름!
stack_containers.cpp
#include <stack>
#include <vector>
#include <deque>
using namespace std;

// 기본 (deque 기반) — 메모리 단편화 가능
stack<int> s1;

// vector 기반 — 캐시 친화적, 정올 추천!
stack<int, vector<int>> s2;

// list 기반 (거의 안 씀)
stack<int, list<int>> s3;

// pair 저장 (인덱스+값 동시에)
stack<pair<int,int>> pstk;
pstk.push({3, 10}); // {인덱스, 값}

STL stack 전체 API

메서드설명복잡도반환값
push(x)top에 원소 삽입O(1)void
pop()top 원소 제거O(1)void ⚠️
top()top 원소 참조O(1)T&
empty()비어있으면 trueO(1)bool
size()원소 개수O(1)size_t
emplace(args)직접 생성 push (C++11)O(1)void
swap(other)두 스택 교환O(1)void
⚠️ pop()은 값을 반환하지 않아!
C++의 pop()은 제거만 해. 값을 꺼내려면 반드시 top()으로 먼저 읽고 pop()으로 제거해야 해!
int v = stk.top(); stk.pop(); — 이 패턴을 외워!

STL stack 기본 사용법

include부터 반복 패턴까지 — 정올에서 바로 쓰는 코드!

stack_basic.cpp — 완전 예제
#include <bits/stdc++.h>  // 정올: 헤더 한 줄로 끝!
using namespace std;

int main() {
    // ① 선언 — vector 기반이 빠름
    stack<int, vector<int>> stk;

    // ② push — O(1)
    stk.push(10);
    stk.push(20);
    stk.push(30);

    // ③ top + pop 패턴
    while (!stk.empty()) {
        cout << stk.top() << " ";  // 30 20 10
        stk.pop();
    }

    // ④ pair 저장 (인덱스, 값 쌍)
    stack<pair<int,int>> ps;
    ps.push({0, 5});   // {인덱스=0, 값=5}
    ps.push({1, 3});
    auto [idx, val] = ps.top();  // C++17 구조적 바인딩
    cout << idx << " " << val;  // 1 3

    return 0;
}

괄호 검사 — STL 활용

bracket.cpp
bool isValid(string s) {
    stack<char> stk;
    for (char c : s) {
        if (c=='('||c=='['||c=='{')
            stk.push(c);
        else {
            if (stk.empty()) return false;
            char t = stk.top(); stk.pop();
            if (c==')' && t!='(') return false;
            if (c==']' && t!='[') return false;
            if (c=='}' && t!='{') return false;
        }
    }
    return stk.empty();
}

스택으로 문자열 역순

reverse.cpp
string reverseStr(string s) {
    stack<char> stk;
    for (char c : s)
        stk.push(c);
    string res;
    while (!stk.empty()) {
        res += stk.top();
        stk.pop();
    }
    return res;
}
// "hello" → "olleh"
정올 꿀팁 — emplace vs push
push({1, 2})는 pair를 먼저 만들고 복사. emplace(1, 2)는 스택 안에서 직접 생성해서 약간 더 빨라.
성능이 민감한 문제라면 emplace 사용을 고려해봐!

단조 스택 (Monotonic Stack)

STL stack으로 NGE(오큰수) 패턴 완성!

단조 스택이란? 스택 안의 원소가 항상 단조 증가 또는 단조 감소 순서를 유지하는 스택.
오큰수(NGE) = 각 원소의 오른쪽에서 자신보다 처음으로 큰 원소.
nge_stl.cpp — 오큰수 (Next Greater Element)
#include <bits/stdc++.h>
using namespace std;

int main() {
    int n; cin >> n;
    vector<int> a(n), ans(n, -1);
    for (int& x : a) cin >> x;

    // STL stack에 인덱스 저장 (단조 감소 유지)
    stack<int, vector<int>> stk;

    for (int i = 0; i < n; i++) {
        // 현재 값이 top보다 크면 → top이 찾던 오큰수!
        while (!stk.empty() && a[stk.top()] < a[i]) {
            ans[stk.top()] = a[i];
            stk.pop();
        }
        stk.push(i);
    }
    // 스택에 남은 것 → 오큰수 없음 (-1)
    for (int x : ans) cout << x << " ";
    return 0;
}
// 입력: 3 5 2 7 / 출력: 5 7 7 -1

NSE — 오왼쪽수 (Next Smaller)

nse_stl.cpp
vector<int> nse(vector<int>& a) {
    int n = a.size();
    vector<int> ans(n, -1);
    stack<int> stk;
    for (int i = 0; i < n; i++) {
        while (!stk.empty() &&
               a[stk.top()] > a[i]) {
            ans[stk.top()] = a[i];
            stk.pop();
        }
        stk.push(i);
    }
    return ans;
}

왼쪽 방향 오큰수 (Previous Greater)

pge_stl.cpp
vector<int> pge(vector<int>& a) {
    int n = a.size();
    vector<int> ans(n, -1);
    stack<int> stk;
    for (int i = 0; i < n; i++) {
        while (!stk.empty() &&
               a[stk.top()] <= a[i])
            stk.pop();
        if (!stk.empty())
            ans[i] = stk.top();
        stk.push(i);
    }
    return ans;
}
단조 스택 비교표
단조 감소 스택 (pop 조건: a[top] < a[i]) → 오큰수(NGE)
단조 증가 스택 (pop 조건: a[top] > a[i]) → 오작은수(NSE), 히스토그램
역방향 순회 → 이전 큰수(PGE), 이전 작은수(PSE)

히스토그램 최대 직사각형

STL stack으로 O(n) 풀이!

문제: 너비 1인 막대들의 히스토그램에서 만들 수 있는 최대 직사각형 넓이를 구하라.
단조 증가 스택으로 O(n)에 해결!
histogram_stl.cpp — 완전 풀이
#include <bits/stdc++.h>
using namespace std;

long long largestRect(vector<int>& h) {
    int n = h.size();
    h.push_back(0); // 센티널 추가!

    // STL stack — 단조 증가 (인덱스 저장)
    stack<int, vector<int>> stk;
    long long maxA = 0;

    for (int i = 0; i <= n; i++) {
        while (!stk.empty() && h[stk.top()] > h[i]) {
            int idx = stk.top(); stk.pop();
            int w = stk.empty() ? i : i - stk.top() - 1;
            maxA = max(maxA, (long long)h[idx] * w);
        }
        stk.push(i);
    }
    h.pop_back(); // 센티널 제거
    return maxA;
}

int main() {
    int n; cin >> n;
    vector<int> h(n);
    for (int& x : h) cin >> x;
    cout << largestRect(h) << "\n";
}

너비 계산 공식 이해

상황너비 공식의미
스택이 비어있음w = i왼쪽 끝까지 전부
스택에 원소 있음w = i - stk.top() - 1두 경계 사이
long long 필수! h[i]×w 최대값이 10^10을 넘을 수 있어. int 오버플로우 주의!

스택 DFS (STL stack 활용)

재귀 대신 STL stack으로 명시적 DFS!

왜 스택 DFS? 재귀 DFS는 시스템 콜 스택을 쓰기 때문에 깊이 제한이 있어.
STL stack을 쓰면 스택 오버플로우 없이 깊은 그래프도 탐색 가능!
dfs_stl.cpp — 인접 행렬
#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1001;

vector<int> adj[MAXN];
bool visited[MAXN];

void dfs_stack(int start) {
    stack<int, vector<int>> stk;
    stk.push(start);

    while (!stk.empty()) {
        int v = stk.top(); stk.pop();
        if (visited[v]) continue;
        visited[v] = true;
        cout << v << " ";
        // 역순 push → 오름차순 방문
        for (int i = adj[v].size()-1; i >= 0; i--)
            if (!visited[adj[v][i]])
                stk.push(adj[v][i]);
    }
}

int main() {
    int n, m; cin >> n >> m;
    for (int i=0; i<m; i++) {
        int u, v; cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    for (int v=1; v<=n; v++)
        sort(adj[v].begin(), adj[v].end());
    dfs_stack(1);
}

투 스택 (Two Stacks)

두 가지 패턴 — 공유 배열 & 두 스택으로 큐!

① 두 스택으로 큐 구현
② 공유 배열 두 스택
두 스택으로 큐 (inbox + outbox)
enqueue → inbox에 push / dequeue → outbox가 비면 inbox를 전부 이동 후 pop
분할 상환 O(1)! 각 원소는 최대 3번만 처리돼.
two_stack_queue_stl.cpp
#include <bits/stdc++.h>
using namespace std;

struct StackQueue {
    stack<int> inbox, outbox;

    void enqueue(int x) {
        inbox.push(x);
    }

    int dequeue() {
        if (outbox.empty()) {
            while (!inbox.empty()) {
                outbox.push(inbox.top());
                inbox.pop();
            }
        }
        if (outbox.empty()) return -1; // empty
        int v = outbox.top(); outbox.pop(); return v;
    }

    int front() {
        if (outbox.empty() && inbox.empty()) return -1;
        if (!outbox.empty()) return outbox.top();
        // inbox bottom이 실제 front
        stack<int> tmp = inbox;
        while (tmp.size() > 1) tmp.pop();
        return tmp.top();
    }
    bool empty() { return inbox.empty() && outbox.empty(); }
    int  size()  { return inbox.size() + outbox.size(); }
};
공유 배열 두 스택
배열 하나를 Stack A(왼쪽→)와 Stack B(←오른쪽)가 나눠 써. 공간 낭비 없음!
STL stack은 독립적이므로, 공유 배열 패턴은 직접 구현하거나 pair 활용.
shared_array_two_stacks.cpp
#include <bits/stdc++.h>
using namespace std;
#define MAX 100

struct TwoStacks {
    int arr[MAX];
    int top1 = -1;       // A: 왼쪽→
    int top2 = MAX;      // B: ←오른쪽

    bool full()  { return top1 + 1 == top2; }
    bool emptyA(){ return top1 == -1; }
    bool emptyB(){ return top2 == MAX; }

    void pushA(int x) { if(!full()) arr[++top1]=x; }
    void pushB(int x) { if(!full()) arr[--top2]=x; }
    int  popA()  { return emptyA()?-1:arr[top1--]; }
    int  popB()  { return emptyB()?-1:arr[top2++]; }
};

최소 스택 (Min Stack)

STL stack 두 개로 getMin() O(1)!

문제: push, pop, top 외에 현재 스택의 최솟값을 O(1)에 반환하는 getMin()을 구현하라.
min_stack_stl.cpp — 보조 스택 방법 (정올 추천)
#include <bits/stdc++.h>
using namespace std;

class MinStack {
    stack<int> stk;     // 메인 스택
    stack<int> minStk;  // 각 단계의 최솟값

public:
    void push(int x) {
        stk.push(x);
        int newMin = minStk.empty() ? x : min(x, minStk.top());
        minStk.push(newMin);
    }

    int pop() {
        int v = stk.top();
        stk.pop(); minStk.pop(); // 항상 같이!
        return v;
    }

    int getMin() { return minStk.top(); } // O(1)!
    int top()    { return stk.top(); }
    bool empty() { return stk.empty(); }
    int size()   { return stk.size(); }
};

쌍(pair) 방법 — 스택 하나로!

min_stack_pair.cpp
class MinStackPair {
    stack<pair<int,int>> stk; // {값, 현재최솟값}
public:
    void push(int x) {
        int curMin = stk.empty() ? x : min(x, stk.top().second);
        stk.push({x, curMin});
    }
    int pop()    { int v=stk.top().first; stk.pop(); return v; }
    int getMin() { return stk.top().second; }
    int top()    { return stk.top().first; }
};
