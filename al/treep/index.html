1

연결 리스트 기반 트리의 핵심

노드의 값char data;
왼쪽 자식 주소struct Node *left;
오른쪽 자식 주소struct Node *right;

배열 방식에서는 자식의 위치를 인덱스로 계산하거나 자식 번호를 저장했습니다. 포인터 방식에서는 자식 노드가 메모리 어디에 있는지 그 주소를 직접 저장합니다.

중요: 자식이 없을 때는 포인터에 NULL을 저장합니다.
2

노드 구조체 선언

자기 참조 구조체
typedef struct Node
{
    char data;

    struct Node *left;
    struct Node *right;
} Node;

왜 struct Node *를 사용할까요?

구조체 안에 같은 구조체의 완전한 객체를 넣으면 끝없이 반복됩니다.

대신 크기가 일정한 주소값을 저장하는 포인터를 넣습니다.

Node *left는 왼쪽 자식 노드를 가리키는 화살표입니다.

malloc으로 노드 생성하기

createNode 함수
Node *createNode(char data)
{
    Node *newNode;

    newNode = (Node *)malloc(sizeof(Node));

    if (newNode == NULL)
    {
        printf("메모리 할당 실패\n");
        exit(1);
    }

    newNode->data = data;
    newNode->left = NULL;
    newNode->right = NULL;

    return newNode;
}

코드 흐름

1. malloc으로 Node 크기만큼 공간 확보
2. 할당 실패 여부 확인
3. 데이터 저장
4. 자식 포인터를 NULL로 초기화
5. 새 노드 주소 반환

동적 메모리 해제하기

왜 후위순회 방식으로 해제할까요?

부모를 먼저 지우면 자식 주소를 잃어버릴 수 있습니다.

왼쪽 자식 → 오른쪽 자식 → 부모 순으로 해제합니다.
void freeTree(Node *root)
{
    if (root == NULL)
        return;

    freeTree(root->left);
    freeTree(root->right);

    free(root);
}

전체 C언어 예제

생성 → 연결 → 순회 → 해제
#include <stdio.h>
#include <stdlib.h>

typedef struct Node
{
    char data;
    struct Node *left;
    struct Node *right;
} Node;

Node *createNode(char data)
{
    Node *newNode = (Node *)malloc(sizeof(Node));

    if (newNode == NULL)
    {
        printf("메모리 할당 실패\n");
        exit(1);
    }

    newNode->data = data;
    newNode->left = NULL;
    newNode->right = NULL;

    return newNode;
}

void preorder(Node *root)
{
    if (root == NULL)
        return;

    printf("%c ", root->data);
    preorder(root->left);
    preorder(root->right);
}

void freeTree(Node *root)
{
    if (root == NULL)
        return;

    freeTree(root->left);
    freeTree(root->right);
    free(root);
}

int main(void)
{
    Node *root = createNode('A');

    root->left = createNode('B');
    root->right = createNode('C');

    root->left->left = createNode('D');
    root->left->right = createNode('E');

    printf("전위순회: ");
    preorder(root);

    freeTree(root);

    return 0;
}
