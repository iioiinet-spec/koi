STL deque — 양방향 큐

세영코딩연구소 · 정올 준비 | Double-Ended Queue — 앞뒤 모두 O(1)!

deque(덱)란? 앞(front)과 뒤(back) 양쪽에서 삽입·삭제가 모두 O(1)인 자료구조.
내부적으로 청크(chunk) 단위 메모리를 사용해서 vector처럼 인덱스 접근도 O(1)이야!

deque_basic.cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    deque<int> dq;                // 선언

    // 앞뒤 모두 삽입
    dq.push_back(10);             // [10]
    dq.push_back(20);             // [10, 20]
    dq.push_front(5);             // [5, 10, 20]
    dq.push_front(1);             // [1, 5, 10, 20]

    cout << dq.front() << "
";  // 1
    cout << dq.back()  << "
";  // 20
    cout << dq[2]     << "
";  // 10 — 인덱스 접근 O(1)!

    // 앞뒤 모두 제거
    dq.pop_front();               // [5, 10, 20]
    dq.pop_back();                // [5, 10]

    // range-based for
    for (int x : dq) cout << x << " ";  // 5 10
    return 0;
}

슬라이딩 윈도우
0-1 BFS
deque as Stack+Queue
슬라이딩 윈도우 최솟값: 크기 k 창이 이동하면서 각 창의 최솟값을 O(n)에!
덱에 인덱스를 저장, 앞은 창 밖이면 제거, 뒤는 단조 증가 유지.
sliding_window_stl.cpp
vector<int> slidingMin(vector<int>& a, int k) {
    int n = a.size();
    vector<int> res;
    deque<int> dq;  // 인덱스 저장
    for (int i = 0; i < n; i++) {
        // ① 창 밖으로 벗어난 인덱스 제거
        while (!dq.empty() && dq.front() <= i - k)
            dq.pop_front();
        // ② 단조 증가 유지 (뒤에서 현재보다 큰 값 제거)
        while (!dq.empty() && a[dq.back()] >= a[i])
            dq.pop_back();
        dq.push_back(i);
        if (i >= k - 1) res.push_back(a[dq.front()]);
    }
    return res;
}
// [3,1,2,5,4,2,6,3] k=3 → [1,1,2,2,2,2,3]
0-1 BFS: 비용이 0 또는 1인 그래프 최단경로를 O(V+E)에!
비용 0 → push_front (현재 레벨 유지) / 비용 1 → push_back (다음 레벨)
zero_one_bfs_stl.cpp
void bfs01(int src, int n) {
    vector<int> dist(n, INT_MAX);
    deque<int> dq;
    dist[src] = 0;
    dq.push_back(src);
    while (!dq.empty()) {
        int v = dq.front(); dq.pop_front();
        for (auto [u, w] : adj[v]) {
            int nd = dist[v] + w;
            if (nd < dist[u]) {
                dist[u] = nd;
                if (w == 0) dq.push_front(u); // 비용 0 → 앞!
                else        dq.push_back(u);  // 비용 1 → 뒤!
            }
        }
    }
}

STL priority_queue

Max-Heap / Min-Heap — O(log n) push/pop, O(1) top!

priority_queue<T, Container, Compare>
기본값: Max-Heap (가장 큰 값이 top). Min-Heap은 greater<T> 사용.
priority_queue_all.cpp
#include <bits/stdc++.h>
using namespace std;
int main() {
    // Max-Heap (기본) — 가장 큰 값 top
    priority_queue<int> maxPQ;
    maxPQ.push(3); maxPQ.push(1); maxPQ.push(7); maxPQ.push(5);
    cout << maxPQ.top() << "
"; // 7
    maxPQ.pop();
    cout << maxPQ.top() << "
"; // 5

    // Min-Heap — 가장 작은 값 top
    priority_queue<int, vector<int>, greater<int>> minPQ;
    minPQ.push(3); minPQ.push(1); minPQ.push(7);
    cout << minPQ.top() << "
"; // 1

    // pair 저장 — {우선순위, 데이터} 패턴 (다익스트라!)
    priority_queue<pair<int,int>, vector<pair<int,int>>,
                  greater<pair<int,int>>> dijkPQ;
    dijkPQ.push({0, 1}); // {거리=0, 노드=1}
    dijkPQ.push({3, 2});
    auto [d, v] = dijkPQ.top(); // d=0, v=1

    // 배열로 초기화 (O(n) heapify!)
    vector<int> arr = {5,2,8,1,9};
    priority_queue<int> pq(arr.begin(), arr.end());
    cout << pq.top() << "
"; // 9
    return 0;
}

다익스트라 — priority_queue 실전 패턴

dijkstra_pq.cpp
using pii = pair<int,int>;
vector<pii> adj[MAXN];
int dist[MAXN];

void dijkstra(int src) {
    fill(dist, dist+MAXN, INT_MAX);
    priority_queue<pii, vector<pii>, greater<pii>> pq;
    dist[src] = 0; pq.push({0, src});
    while (!pq.empty()) {
        auto [d, v] = pq.top(); pq.pop();
        if (d > dist[v]) continue; // 구식값 무시!
        for (auto [u, w] : adj[v]) {
            if (dist[v]+w < dist[u]) {
                dist[u] = dist[v]+w;
                pq.push({dist[u], u});
            }
        }
    }
}

방법 1: 구조체에 operator<를 오버로딩. Max-Heap 기준으로 동작.
custom_pq_operator.cpp
struct Task {
    int cost, id;
    // operator< 오버로딩 — priority_queue는 Max-Heap!
    bool operator<(const Task& o) const {
        if (cost != o.cost) return cost > o.cost; // 비용 낮은 게 우선
        return id > o.id;  // 같으면 id 낮은 게 우선
    }
};

priority_queue<Task> pq;
pq.push({3, 1});  // {cost=3, id=1}
pq.push({1, 2});  // {cost=1, id=2}
pq.push({3, 0});  // {cost=3, id=0}
while(!pq.empty()){
    cout << pq.top().cost << " " << pq.top().id << "
";
    pq.pop();
}
// 출력: 1 2 / 3 0 / 3 1
주의! priority_queue는 Max-Heap이야. operator<에서 "큰 것"이 top이 되도록 반전해서 써야 해!
방법 2: 별도 Comparator 구조체 작성. 가장 명시적이고 안전한 방법!
custom_pq_struct_cmp.cpp
struct Task { int cost, id, priority; };

struct Cmp {
    bool operator()(const Task& a, const Task& b) const {
        // true면 b가 우선 (a를 b 뒤로 밀어냄)
        if (a.cost != b.cost) return a.cost > b.cost;
        if (a.priority != b.priority) return a.priority < b.priority;
        return a.id > b.id;
    }
};

priority_queue<Task, vector<Task>, Cmp> pq;
pq.push({3, 1, 5}); // {cost, id, priority}
pq.push({1, 2, 3});
pq.push({3, 0, 8});
방법 3: C++14 lambda — 간결하지만 타입 추론이 필요해서 decltype 또는 function 사용.
custom_pq_lambda.cpp
struct Task { int cost, id; };

// C++14: auto + decltype
auto cmp = [](const Task& a, const Task& b) {
    return a.cost > b.cost; // 비용 낮은 게 우선
};
priority_queue<Task, vector<Task>, decltype(cmp)> pq(cmp);

// C++11: function<> 버전
using CmpFn = function<bool(const Task&, const Task&)>;
CmpFn cmp2 = [](const Task& a, const Task& b){
    return a.cost > b.cost;
};
priority_queue<Task, vector<Task>, CmpFn> pq2(cmp2);
실전: 작업 스케줄링 — 마감기한이 빠른 것 우선, 같으면 보상이 큰 것 우선!
job_scheduling.cpp
struct Job {
    int deadline, reward, id;
    bool operator<(const Job& o) const {
        if (deadline != o.deadline) return deadline > o.deadline;
        return reward < o.reward; // 보상 큰 게 우선
    }
};

priority_queue<Job> pq;
pq.push({3, 50, 1}); // 마감3, 보상50
pq.push({1, 80, 2}); // 마감1, 보상80
pq.push({3, 70, 3}); // 마감3, 보상70

while(!pq.empty()) {
    auto j = pq.top(); pq.pop();
    cout << "ID:" << j.id << " d=" << j.deadline
         << " r=" << j.reward << "
";
}
// 마감1→id2(80), 마감3→id3(70), 마감3→id1(50)

circular_queue_optimized.cpp
#include <bits/stdc++.h>
using namespace std;
#define QMAX 1000001

// 원형 큐 — 고정 배열 + 모듈러
int cq[QMAX];
int head = 0, tail = 0;

// enqueue — O(1), 캐시 친화적!
inline void push(int x) {
    cq[tail] = x;
    tail = (tail + 1) % QMAX;
}

// dequeue — O(1)
inline int pop() {
    int v = cq[head];
    head = (head + 1) % QMAX;
    return v;
}

inline bool empty() { return head == tail; }
inline int  size()  { return (tail - head + QMAX) % QMAX; }
inline int  front() { return cq[head]; }

int main() {
    push(1); push(2); push(3);
    cout << pop() << "
"; // 1
    push(4); push(5);
    cout << size() << "
"; // 4
}
BFS 최적화: pair 큐 대신 배열 두 개 + 인덱스로 최고 성능!
bfs_optimized.cpp
#define QMAX 1000001
// pair를 두 배열로 분리 → 메모리 연속성 향상
int qr[QMAX], qc[QMAX];
int head = 0, tail = 0;

inline void enq(int r, int c) {
    qr[tail] = r; qc[tail] = c;
    tail = (tail + 1) % QMAX;
}

// BFS
int dr[]={-1,1,0,0}, dc[]={0,0,-1,1};
enq(sr, sc); dist[sr][sc] = 0;
while (head != tail) {
    int r = qr[head], c = qc[head];
    head = (head + 1) % QMAX;
    for (int d = 0; d < 4; d++) {
        int nr=r+dr[d], nc=c+dc[d];
        if(...) { dist[nr][nc]=dist[r][c]+1; enq(nr,nc); }
    }
}

deque 선택 기준

  • 앞뒤 모두 삽입·삭제
  • 슬라이딩 윈도우
  • 0-1 BFS 구현
  • 인덱스 접근 필요

priority_queue 선택 기준

  • 항상 최대/최솟값 꺼내기
  • 다익스트라 최단 경로
  • K번째 최솟값
  • 힙 정렬 O(n log n)

원형 큐 선택 기준

  • BFS 시간 초과 경계
  • 고정 크기 큐 필요
  • 메모리 재사용 필요
  • inline 최적화 원할 때
