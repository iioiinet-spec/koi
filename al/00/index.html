1. 기본 스택

1. 개념

스택은 마지막에 넣은 것이 가장 먼저 나오는 자료구조입니다.

이것을 LIFO라고 합니다.

명령뜻
push스택에 넣기
pop맨 위 값 빼기
top맨 위 값 확인
empty비었는지 확인

2. 직접 실습

스택이 비어 있습니다.

3. 기본 C 코드

#include <stdio.h>
#define MAX 100000
int stack[MAX];
int topIndex = -1;
void push(int x){ stack[++topIndex] = x; }
int pop(){ return stack[topIndex--]; }
int top(){ return stack[topIndex]; }
int empty(){ return topIndex == -1; }
int size(){ return topIndex + 1; }
int main(){
    push(10); push(20);
    printf("%d\n", top());
    printf("%d\n", pop());
    return 0;
}

2. 괄호 매칭

1. 개념

여는 괄호가 나오면 넣고, 닫는 괄호가 나오면 맨 위 여는 괄호와 짝이 맞는지 확인합니다.

닫는 괄호가 나왔는데 스택이 비어 있거나 짝이 다르면 잘못된 괄호입니다.

2. 실습

3. C 코드

#include <stdio.h>
char st[100005];
int topIndex = -1;
int pair(char open, char close){
    if(open=='(' && close==')') return 1;
    if(open=='{' && close=='}') return 1;
    if(open=='[' && close==']') return 1;
    return 0;
}
int main(){
    char s[100005]; scanf("%s", s);
    int ok = 1;
    for(int i=0; s[i]; i++){
        if(s[i]=='(' || s[i]=='{' || s[i]=='[') st[++topIndex]=s[i];
        else{
            if(topIndex == -1 || !pair(st[topIndex], s[i])){ ok=0; break; }
            topIndex--;
        }
    }
    if(topIndex != -1) ok=0;
    printf(ok ? "YES\n" : "NO\n");
    return 0;
}

3. 후위표기식

1. 개념

숫자는 스택에 넣고, 연산자가 나오면 숫자 2개를 꺼내 계산한 뒤 다시 넣습니다.

중위표기식후위표기식
3 + 43 4 +
3 + 4 × 53 4 5 × +

2. 실습

3. C 코드

#include <stdio.h>
#include <ctype.h>
int st[100005], topIndex=-1;
int main(){
    char s[100005]; scanf("%s", s);
    for(int i=0; s[i]; i++){
        if(isdigit(s[i])) st[++topIndex] = s[i]-'0';
        else{
            int b=st[topIndex--], a=st[topIndex--];
            if(s[i]=='+') st[++topIndex]=a+b;
            if(s[i]=='-') st[++topIndex]=a-b;
            if(s[i]=='*') st[++topIndex]=a*b;
            if(s[i]=='/') st[++topIndex]=a/b;
        }
    }
    printf("%d\n", st[topIndex]);
    return 0;
}

4. DFS 스택

1. 개념

DFS는 깊이 우선 탐색입니다. 재귀 대신 스택으로 만들 수 있습니다.

시작 정점을 스택에 넣는다.
while (스택이 비어 있지 않다) {
    하나를 꺼낸다.
    아직 방문하지 않았다면 방문 처리한다.
    연결된 다음 정점들을 스택에 넣는다.
}

2. C 코드

#include <stdio.h>
#define MAXN 1005
int graph[MAXN][MAXN], visited[MAXN], st[MAXN];
int topIndex=-1, N, M;
void push(int x){ st[++topIndex]=x; }
int pop(){ return st[topIndex--]; }
int empty(){ return topIndex==-1; }
int main(){
    scanf("%d %d", &N, &M);
    for(int i=0;i<M;i++){
        int a,b; scanf("%d %d", &a, &b);
        graph[a][b]=graph[b][a]=1;
    }
    push(1);
    while(!empty()){
        int now=pop();
        if(visited[now]) continue;
        visited[now]=1;
        printf("%d ", now);
        for(int next=N; next>=1; next--)
            if(graph[now][next] && !visited[next]) push(next);
    }
    return 0;
}

5. 모노톤 스택

1. 개념

스택 안의 값이 항상 증가하거나 감소하도록 관리하는 스택입니다. 가까운 큰 값, 가까운 작은 값을 빠르게 찾습니다.

각 원소는 스택에 한 번 들어가고 한 번만 나옵니다. 전체 시간복잡도는 O(N)입니다.

2. C 기본 코드

#include <stdio.h>
int a[100005], st[100005], topIndex=-1;
int main(){
    int N; scanf("%d", &N);
    for(int i=0;i<N;i++){
        scanf("%d", &a[i]);
        while(topIndex!=-1 && st[topIndex] > a[i]) topIndex--;
        st[++topIndex]=a[i];
    }
    return 0;
}

6. 오큰수

1. 개념

어떤 수의 오른쪽에 있으면서 가장 먼저 만나는 큰 수입니다.

2. C 코드

#include <stdio.h>
#define MAXN 1000005
int a[MAXN], ans[MAXN], st[MAXN], topIndex=-1;
int main(){
    int N; scanf("%d", &N);
    for(int i=0;i<N;i++){ scanf("%d", &a[i]); ans[i]=-1; }
    for(int i=0;i<N;i++){
        while(topIndex!=-1 && a[st[topIndex]] < a[i]){
            ans[st[topIndex]]=a[i];
            topIndex--;
        }
        st[++topIndex]=i;
    }
    for(int i=0;i<N;i++) printf("%d ", ans[i]);
    return 0;
}

7. 히스토그램 최대 직사각형

1. 개념

막대그래프에서 만들 수 있는 가장 큰 직사각형 넓이를 찾습니다. 현재 막대보다 낮은 막대를 만나면 이전 높은 막대의 넓이를 계산합니다.

2. C 코드

#include <stdio.h>
#define MAXN 100005
long long h[MAXN];
int st[MAXN];
int main(){
    int N; scanf("%d", &N);
    for(int i=0;i<N;i++) scanf("%lld", &h[i]);
    h[N]=0;
    int topIndex=-1;
    long long answer=0;
    for(int i=0;i<=N;i++){
        while(topIndex!=-1 && h[st[topIndex]] > h[i]){
            int idx=st[topIndex--];
            long long height=h[idx];
            int left = (topIndex!=-1 ? st[topIndex] : -1);
            long long width = i-left-1;
            long long area = height*width;
            if(answer < area) answer=area;
        }
        st[++topIndex]=i;
    }
    printf("%lld\n", answer);
    return 0;
}

8. 문자열 폭발

1. 개념

문자를 하나씩 스택에 넣다가, 스택의 끝부분이 폭발 문자열과 같아지면 제거합니다.

2. C 코드

#include <stdio.h>
#include <string.h>
char s[1000005], bomb[40], st[1000005];
int main(){
    scanf("%s", s); scanf("%s", bomb);
    int len=strlen(s), blen=strlen(bomb), topIndex=0;
    for(int i=0;i<len;i++){
        st[topIndex++]=s[i];
        if(topIndex >= blen){
            int same=1;
            for(int j=0;j<blen;j++)
                if(st[topIndex-blen+j] != bomb[j]) same=0;
            if(same) topIndex -= blen;
        }
    }
    if(topIndex==0) printf("FRULA\n");
    else{
        for(int i=0;i<topIndex;i++) printf("%c", st[i]);
        printf("\n");
    }
    return 0;
}

9. 최소 스택

1. 개념

현재 스택 안의 최솟값을 바로 알 수 있는 스택입니다. 값 스택과 최소값 스택을 함께 사용합니다.

2. C 코드

#include <stdio.h>
#define MAXN 100005
int st[MAXN], minSt[MAXN];
int topIndex=-1, minTop=-1;
void push(int x){
    st[++topIndex]=x;
    if(minTop==-1 || x <= minSt[minTop]) minSt[++minTop]=x;
}
void pop(){
    if(st[topIndex] == minSt[minTop]) minTop--;
    topIndex--;
}
int getMin(){ return minSt[minTop]; }
int main(){
    push(5); push(3); push(7);
    printf("%d\n", getMin());
    pop();
    printf("%d\n", getMin());
    return 0;
}

10. 카테시안 트리

1. 개념

배열을 트리로 바꾸는 고급 자료구조입니다. 배열의 순서는 중위순회 순서와 같고, 부모는 자식보다 작거나 큽니다.

카테시안 트리는 모노톤 스택으로 O(N)에 만들 수 있습니다.

2. C 코드

#include <stdio.h>
#define MAXN 100005
int a[MAXN], leftChild[MAXN], rightChild[MAXN], parent[MAXN], st[MAXN];
int main(){
    int N; scanf("%d", &N);
    for(int i=1;i<=N;i++) scanf("%d", &a[i]);
    int topIndex=0;
    for(int i=1;i<=N;i++){
        int last=0;
        while(topIndex>0 && a[st[topIndex]] > a[i]){
            last=st[topIndex];
            topIndex--;
        }
        if(topIndex>0){
            rightChild[st[topIndex]]=i;
            parent[i]=st[topIndex];
        }
        if(last!=0){
            leftChild[i]=last;
            parent[last]=i;
        }
        st[++topIndex]=i;
    }
    int root=1;
    while(parent[root]!=0) root=parent[root];
    printf("root = %d\n", root);
    return 0;
}
