10. 정답 C 코드

요청하신 대로 전체 정답 코드는 강의자료의 가장 마지막에 배치했습니다.

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_LEVEL 64

static int currentK;
static long long queryLeft;
static long long queryRight;
static int highestLevel;

typedef struct {
    long long *left;
    long long *right;

    int leftSize;
    int rightSize;

    long long best;
    long long zeroCount;
} Unit;

static Unit singleUnit[MAX_LEVEL];
static Unit precomputed[MAX_LEVEL];

static long long maximum(
    long long a,
    long long b
)
{
    return (a > b) ? a : b;
}

static Unit emptyUnit(void)
{
    Unit result;

    result.left = NULL;
    result.right = NULL;

    result.leftSize = 0;
    result.rightSize = 0;

    result.best = 0;
    result.zeroCount = 0;

    return result;
}

static void freeUnit(Unit *unit)
{
    free(unit->left);
    free(unit->right);

    unit->left = NULL;
    unit->right = NULL;

    unit->leftSize = 0;
    unit->rightSize = 0;
}

static Unit copyUnit(const Unit *source)
{
    Unit result = emptyUnit();

    result.leftSize =
        source->leftSize;

    result.rightSize =
        source->rightSize;

    result.best =
        source->best;

    result.zeroCount =
        source->zeroCount;

    result.left =
        (long long *)malloc(
            sizeof(long long) *
            result.leftSize
        );

    result.right =
        (long long *)malloc(
            sizeof(long long) *
            result.rightSize
        );

    memcpy(
        result.left,
        source->left,
        sizeof(long long) *
        result.leftSize
    );

    memcpy(
        result.right,
        source->right,
        sizeof(long long) *
        result.rightSize
    );

    return result;
}

static Unit makeSingleUnit(int bit)
{
    Unit result = emptyUnit();

    result.zeroCount =
        (bit == 0) ? 1 : 0;

    if (currentK == 0) {
        result.best = bit;
    } else {
        result.best = 1;
    }

    result.leftSize =
        1 +
        (
            currentK > 0 &&
            bit == 0
        );

    result.rightSize =
        result.leftSize;

    result.left =
        (long long *)malloc(
            sizeof(long long) *
            result.leftSize
        );

    result.right =
        (long long *)malloc(
            sizeof(long long) *
            result.rightSize
        );

    result.left[0] = bit;
    result.right[0] = bit;

    if (
        currentK > 0 &&
        bit == 0
    ) {
        result.left[1] = 1;
        result.right[1] = 1;
    }

    return result;
}

static Unit mergeUnit(
    const Unit *x,
    const Unit *y
)
{
    Unit result = emptyUnit();

    result.zeroCount =
        x->zeroCount +
        y->zeroCount;

    result.best =
        maximum(
            x->best,
            y->best
        );

    result.left =
        (long long *)malloc(
            sizeof(long long) *
            (currentK + 1)
        );

    result.right =
        (long long *)malloc(
            sizeof(long long) *
            (currentK + 1)
        );

    result.leftSize =
        x->leftSize;

    for (int i = 0;
         i < x->leftSize;
         i++) {

        result.left[i] =
            x->left[i];
    }

    if (x->zeroCount <= currentK) {

        result.leftSize--;

        for (
            int i = 0;
            i < y->leftSize &&
            result.leftSize <= currentK;
            i++
        ) {
            result.left[
                result.leftSize++
            ] =
                x->best +
                y->left[i];
        }
    }

    result.rightSize =
        y->rightSize;

    for (int i = 0;
         i < y->rightSize;
         i++) {

        result.right[i] =
            y->right[i];
    }

    if (y->zeroCount <= currentK) {

        result.rightSize--;

        for (
            int i = 0;
            i < x->rightSize &&
            result.rightSize <= currentK;
            i++
        ) {
            result.right[
                result.rightSize++
            ] =
                y->best +
                x->right[i];
        }
    }

    int allowed;

    if (result.zeroCount < currentK) {
        allowed =
            (int)result.zeroCount;
    } else {
        allowed =
            currentK;
    }

    int firstLeftZero =
        allowed -
        (y->leftSize - 1);

    if (firstLeftZero < 0) {
        firstLeftZero = 0;
    }

    int lastLeftZero =
        x->rightSize - 1;

    if (lastLeftZero > allowed) {
        lastLeftZero = allowed;
    }

    for (
        int leftZero = firstLeftZero;
        leftZero <= lastLeftZero;
        leftZero++
    ) {
        int rightZero =
            allowed - leftZero;

        long long candidate =
            x->right[leftZero] +
            y->left[rightZero];

        if (candidate > result.best) {
            result.best = candidate;
        }
    }

    return result;
}

static int trailingZeroCount(
    unsigned long long value
)
{
    return __builtin_ctzll(value);
}

static void prepareUnits(
    unsigned long long N
)
{
    for (int level = 0;
         level <= highestLevel;
         level++) {

        int bitPosition =
            highestLevel - level;

        int bit =
            (int)(
                (N >> bitPosition) & 1ULL
            );

        singleUnit[level] =
            makeSingleUnit(bit);
    }

    precomputed[0] =
        copyUnit(
            &singleUnit[0]
        );

    for (int level = 1;
         level <= highestLevel;
         level++) {

        Unit temporary =
            mergeUnit(
                &precomputed[level - 1],
                &singleUnit[level]
            );

        precomputed[level] =
            mergeUnit(
                &temporary,
                &precomputed[level - 1]
            );

        freeUnit(&temporary);
    }
}

static void clearUnits(void)
{
    for (int level = 0;
         level <= highestLevel;
         level++) {

        freeUnit(
            &singleUnit[level]
        );

        freeUnit(
            &precomputed[level]
        );
    }
}

static Unit divideAndConquer(
    long long start,
    long long end
)
{
    long long middle =
        start +
        (end - start) / 2;

    int currentLevel =
        trailingZeroCount(
            (unsigned long long)middle
        );

    if (
        queryLeft <= start &&
        end <= queryRight
    ) {
        return copyUnit(
            &precomputed[currentLevel]
        );
    }

    if (queryRight < middle) {
        return divideAndConquer(
            start,
            middle - 1
        );
    }

    if (queryLeft > middle) {
        return divideAndConquer(
            middle + 1,
            end
        );
    }

    if (queryLeft == middle) {
        Unit rightPart =
            divideAndConquer(
                middle + 1,
                end
            );

        Unit result =
            mergeUnit(
                &singleUnit[currentLevel],
                &rightPart
            );

        freeUnit(&rightPart);

        return result;
    }

    Unit leftPart =
        divideAndConquer(
            start,
            middle - 1
        );

    if (queryRight == middle) {
        Unit result =
            mergeUnit(
                &leftPart,
                &singleUnit[currentLevel]
            );

        freeUnit(&leftPart);

        return result;
    }

    Unit rightPart =
        divideAndConquer(
            middle + 1,
            end
        );

    Unit leftAndMiddle =
        mergeUnit(
            &leftPart,
            &singleUnit[currentLevel]
        );

    Unit result =
        mergeUnit(
            &leftAndMiddle,
            &rightPart
        );

    freeUnit(&leftPart);
    freeUnit(&rightPart);
    freeUnit(&leftAndMiddle);

    return result;
}

int main(void)
{
    unsigned long long N;
    int Q;

    scanf(
        "%llu %d",
        &N,
        &Q
    );

    highestLevel =
        63 -
        __builtin_clzll(N);

    unsigned long long totalLength =
        (
            1ULL <<
            (highestLevel + 1)
        ) - 1ULL;

    for (int query = 0;
         query < Q;
         query++) {

        scanf(
            "%lld %lld %d",
            &queryLeft,
            &queryRight,
            &currentK
        );

        prepareUnits(N);

        Unit answer;

        if (queryLeft == queryRight) {
            int level =
                trailingZeroCount(
                    (unsigned long long)
                    queryLeft
                );

            answer =
                copyUnit(
                    &singleUnit[level]
                );
        } else {
            answer =
                divideAndConquer(
                    1,
                    (long long)totalLength
                );
        }

        printf(
            "%lld\n",
            answer.best
        );

        freeUnit(&answer);
        clearUnits();
    }

    return 0;
}
