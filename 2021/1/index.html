6. 핵심 점화식 이해

store[i][j] = max(output[i-1][j-1], output[i][j-1], output[i+1][j-1])
과거의 넓은 삼각형 영역
→
이전 열 위 로봇이 요약
+
가운데 로봇이 요약
+
아래 로봇이 요약

왜 가능한가?

이전 열의 세 로봇은 각각 자신까지 도달 가능한 모든 과거 출력값의 최댓값을 이미 저장하고 있습니다.

그래서

현재 로봇은 과거 모든 칸을 다시 볼 필요 없이 세 출력값만 비교하면 됩니다.

7. 완전탐색과 DP 비교

완전탐색

각 로봇이 왼쪽 전체 영역을 다시 검사하므로 매우 느립니다.

DP 한 칸

위·가운데·아래 최대 3개만 비교하므로 O(1)입니다.

전체

M×N칸을 한 번씩 처리하므로 O(MN)입니다.

9. 정답 C 코드

요청하신 대로 정답 코드는 강의자료의 가장 마지막에 배치했습니다.

#include <stdio.h>
#include <stdlib.h>

int main(void)
{
    int M, N;

    if (scanf("%d %d", &M, &N) != 2) {
        return 0;
    }

    char *row =
        (char *)malloc(
            sizeof(char) * (N + 1)
        );

    unsigned char *weight =
        (unsigned char *)malloc(
            sizeof(unsigned char)
            * (size_t)M
            * (size_t)N
        );

    if (row == NULL || weight == NULL) {
        free(row);
        free(weight);
        return 0;
    }

    for (int i = 0; i < M; i++) {
        scanf("%s", row);

        for (int j = 0; j < N; j++) {
            weight[
                (size_t)i * N + j
            ] =
                (unsigned char)(
                    row[j] - '0'
                );
        }
    }

    int *prev =
        (int *)malloc(
            sizeof(int) * M
        );

    int *curr =
        (int *)malloc(
            sizeof(int) * M
        );

    if (prev == NULL || curr == NULL) {
        free(curr);
        free(prev);
        free(weight);
        free(row);
        return 0;
    }

    int answer = 0;

    /*
     * 첫 번째 열
     *
     * 저장 값은 모두 0
     * 출력 값은 가중치와 같다.
     */
    for (int i = 0; i < M; i++) {
        prev[i] =
            weight[(size_t)i * N];
    }

    /*
     * 두 번째 열부터 처리
     */
    for (int j = 1; j < N; j++) {

        for (int i = 0; i < M; i++) {

            /*
             * 이전 열의 가운데 출력값
             */
            int best = prev[i];

            /*
             * 이전 열의 위쪽 출력값
             */
            if (
                i > 0 &&
                prev[i - 1] > best
            ) {
                best = prev[i - 1];
            }

            /*
             * 이전 열의 아래쪽 출력값
             */
            if (
                i + 1 < M &&
                prev[i + 1] > best
            ) {
                best = prev[i + 1];
            }

            /*
             * best는 현재 로봇의 저장 값
             */
            if (best > answer) {
                answer = best;
            }

            /*
             * 현재 로봇의 출력 값
             */
            curr[i] =
                best +
                weight[
                    (size_t)i * N + j
                ];
        }

        /*
         * 배열을 복사하지 않고
         * 포인터만 교환
         */
        int *temp = prev;
        prev = curr;
        curr = temp;
    }

    printf("%d\n", answer);

    free(curr);
    free(prev);
    free(weight);
    free(row);

    return 0;
}
