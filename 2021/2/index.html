1. 문제를 가장 쉽게 이해하기

마을

각 마을은 위치 x와 인구수 a를 가지고 있습니다.

후보 장소 q

모임 장소가 q일 때 모든 사람이 이동한 거리의 합을 구합니다.

핵심 수식

마을 하나의 기여도는 a × |x-q|입니다.

f(q) = Σ aᵢ × |xᵢ - q|

8. 왜 공식이 맞는가?

왼쪽 마을

x < q 이므로 |x-q| = q-x

a(q-x) = aq-ax

모두 더하면 qΣa - Σax

오른쪽 마을

x ≥ q 이므로 |x-q| = x-q

a(x-q) = ax-aq

모두 더하면 Σax - qΣa

11. 정답 C 코드

전체 정답 코드는 요청하신 대로 강의자료 맨 뒤에 배치했습니다.

#include <stdio.h>
#include <stdlib.h>

typedef struct {
    long long population;
    long long position;
} Village;

int compareVillage(
    const void *left,
    const void *right
)
{
    const Village *a =
        (const Village *)left;

    const Village *b =
        (const Village *)right;

    if (a->position < b->position) {
        return -1;
    }

    if (a->position > b->position) {
        return 1;
    }

    return 0;
}

int lowerBound(
    const Village village[],
    int size,
    long long target
)
{
    int left = 0;
    int right = size;

    while (left < right) {
        int middle =
            left + (right - left) / 2;

        if (
            village[middle].position
            < target
        ) {
            left = middle + 1;
        } else {
            right = middle;
        }
    }

    return left;
}

int main(void)
{
    int N;
    int Q;

    scanf("%d %d", &N, &Q);

    Village *village =
        (Village *)malloc(
            sizeof(Village) * N
        );

    for (int i = 0; i < N; i++) {
        scanf(
            "%lld %lld",
            &village[i].population,
            &village[i].position
        );
    }

    qsort(
        village,
        N,
        sizeof(Village),
        compareVillage
    );

    long long *prefixPopulation =
        (long long *)malloc(
            sizeof(long long) *
            (N + 1)
        );

    long long *prefixWeighted =
        (long long *)malloc(
            sizeof(long long) *
            (N + 1)
        );

    prefixPopulation[0] = 0;
    prefixWeighted[0] = 0;

    for (int i = 0; i < N; i++) {
        prefixPopulation[i + 1] =
            prefixPopulation[i]
            + village[i].population;

        prefixWeighted[i + 1] =
            prefixWeighted[i]
            + village[i].population
              * village[i].position;
    }

    for (int query = 0;
         query < Q;
         query++) {

        long long meetingPosition;

        scanf(
            "%lld",
            &meetingPosition
        );

        int boundary =
            lowerBound(
                village,
                N,
                meetingPosition
            );

        long long leftPopulation =
            prefixPopulation[boundary];

        long long leftWeighted =
            prefixWeighted[boundary];

        long long rightPopulation =
            prefixPopulation[N]
            - prefixPopulation[boundary];

        long long rightWeighted =
            prefixWeighted[N]
            - prefixWeighted[boundary];

        long long leftDistance =
            meetingPosition
            * leftPopulation
            - leftWeighted;

        long long rightDistance =
            rightWeighted
            - meetingPosition
              * rightPopulation;

        long long answer =
            leftDistance
            + rightDistance;

        printf("%lld\n", answer);
    }

    free(prefixWeighted);
    free(prefixPopulation);
    free(village);

    return 0;
}
