5. 코드 작성 단계

1단계. 좌표 압축

배열을 복사하고 정렬한 뒤 중복을 제거한다.
각 Ai를 1부터 valueCount까지의 압축 번호로 바꾼다.

2단계. leftGreater 계산

leftGreater[i]
= (i - 1)
- 왼쪽에서 Ai 이하인 원소 수

3단계. rightLess 계산

rightLess[i]
= 오른쪽에서 Ai보다 작은 원소 수

4단계. 새 값의 비용

cost(x)
= 왼쪽에서 x보다 큰 원소 수
+ 오른쪽에서 x보다 작은 원소 수

5단계. 값 v의 이벤트

x = v가 되면: -leftCount[v]
x가 v보다 커지면: +rightCount[v]

6단계. 세그먼트 트리 노드

sum[node]
minimumPrefix[node]

minimumPrefix[node] =
min(
    minimumPrefix[left],
    sum[left] + minimumPrefix[right]
);

6. 정답 C 코드

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_N 300000
#define SEGMENT_SIZE (MAX_N * 4 + 20)

int N;

int A[MAX_N + 1];
int sortedValue[MAX_N + 1];
int rankValue[MAX_N + 1];

int fenwick[MAX_N + 1];

long long leftGreater[MAX_N + 1];
long long rightLess[MAX_N + 1];

int leftCount[MAX_N + 1];
int rightCount[MAX_N + 1];

long long segmentSum[SEGMENT_SIZE];
long long segmentMin[SEGMENT_SIZE];

int compareInt(
    const void *a,
    const void *b
) {
    int valueA = *(const int *)a;
    int valueB = *(const int *)b;

    if (valueA < valueB) {
        return -1;
    }

    if (valueA > valueB) {
        return 1;
    }

    return 0;
}

int getRank(
    int value,
    int valueCount
) {
    int left = 1;
    int right = valueCount;

    while (left <= right) {
        int middle = (left + right) / 2;

        if (sortedValue[middle] == value) {
            return middle;
        }

        if (sortedValue[middle] < value) {
            left = middle + 1;
        } else {
            right = middle - 1;
        }
    }

    return -1;
}

void fenwickAdd(
    int index,
    int valueCount
) {
    while (index <= valueCount) {
        fenwick[index]++;
        index += index & -index;
    }
}

long long fenwickSum(int index) {
    long long result = 0;

    while (index > 0) {
        result += fenwick[index];
        index -= index & -index;
    }

    return result;
}

long long minimumLongLong(
    long long a,
    long long b
) {
    return a < b ? a : b;
}

void pull(int node) {
    int leftChild = node * 2;
    int rightChild = node * 2 + 1;

    segmentSum[node] =
        segmentSum[leftChild] +
        segmentSum[rightChild];

    segmentMin[node] =
        minimumLongLong(
            segmentMin[leftChild],
            segmentSum[leftChild] +
                segmentMin[rightChild]
        );
}

void buildSegmentTree(
    int node,
    int start,
    int end
) {
    if (start == end) {
        segmentSum[node] =
            (long long)rightCount[start] -
            leftCount[start];

        segmentMin[node] =
            -((long long)leftCount[start]);

        return;
    }

    int middle = (start + end) / 2;

    buildSegmentTree(
        node * 2,
        start,
        middle
    );

    buildSegmentTree(
        node * 2 + 1,
        middle + 1,
        end
    );

    pull(node);
}

void updateSegmentTree(
    int node,
    int start,
    int end,
    int position
) {
    if (start == end) {
        segmentSum[node] =
            (long long)rightCount[position] -
            leftCount[position];

        segmentMin[node] =
            -((long long)leftCount[position]);

        return;
    }

    int middle = (start + end) / 2;

    if (position <= middle) {
        updateSegmentTree(
            node * 2,
            start,
            middle,
            position
        );
    } else {
        updateSegmentTree(
            node * 2 + 1,
            middle + 1,
            end,
            position
        );
    }

    pull(node);
}

int main(void) {
    scanf("%d", &N);

    for (int i = 1; i <= N; i++) {
        scanf("%d", &A[i]);
        sortedValue[i] = A[i];
    }

    qsort(
        sortedValue + 1,
        N,
        sizeof(int),
        compareInt
    );

    int valueCount = 0;

    for (int i = 1; i <= N; i++) {
        if (
            i == 1 ||
            sortedValue[i] != sortedValue[i - 1]
        ) {
            sortedValue[++valueCount] =
                sortedValue[i];
        }
    }

    for (int i = 1; i <= N; i++) {
        rankValue[i] =
            getRank(A[i], valueCount);
    }

    long long totalInversions = 0;

    memset(fenwick, 0, sizeof(fenwick));

    for (int i = 1; i <= N; i++) {
        long long lessOrEqual =
            fenwickSum(rankValue[i]);

        leftGreater[i] =
            (long long)(i - 1) -
            lessOrEqual;

        totalInversions += leftGreater[i];

        fenwickAdd(
            rankValue[i],
            valueCount
        );
    }

    memset(fenwick, 0, sizeof(fenwick));

    for (int i = N; i >= 1; i--) {
        rightLess[i] =
            fenwickSum(rankValue[i] - 1);

        fenwickAdd(
            rankValue[i],
            valueCount
        );
    }

    for (int i = 2; i <= N; i++) {
        rightCount[rankValue[i]]++;
    }

    buildSegmentTree(
        1,
        1,
        valueCount
    );

    long long leftSize = 0;

    for (int i = 1; i <= N; i++) {
        long long bestNewCost =
            leftSize + segmentMin[1];

        long long result =
            totalInversions
            - leftGreater[i]
            - rightLess[i]
            + bestNewCost;

        if (i > 1) {
            printf(" ");
        }

        printf("%lld", result);

        if (i < N) {
            int currentRank =
                rankValue[i];

            leftCount[currentRank]++;

            updateSegmentTree(
                1,
                1,
                valueCount,
                currentRank
            );

            int nextRank =
                rankValue[i + 1];

            rightCount[nextRank]--;

            updateSegmentTree(
                1,
                1,
                valueCount,
                nextRank
            );

            leftSize++;
        }
    }

    printf("\n");

    return 0;
}
