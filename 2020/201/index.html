4단계. D/U/R/L 배치 규칙

D: 위쪽 반전본, 아래쪽 원본
U: 위쪽 원본, 아래쪽 반전본
R: 왼쪽 반전본, 오른쪽 원본
L: 왼쪽 원본, 오른쪽 반전본

5. 정답 C 코드

#include <stdio.h>
#define MAX_K 8
#define MAX_SIZE 256
char command[2 * MAX_K];
int paper[MAX_SIZE][MAX_SIZE];
int nextPaper[MAX_SIZE][MAX_SIZE];
int flipVertical(int hole) { return hole ^ 2; }
int flipHorizontal(int hole) { return hole ^ 1; }
int main(void) {
    int k, h;
    scanf("%d", &k);
    for (int i = 0; i < 2 * k; i++) scanf(" %c", &command[i]);
    scanf("%d", &h);
    int rows = 1, cols = 1;
    paper[0][0] = h;
    for (int index = 2 * k - 1; index >= 0; index--) {
        char d = command[index];
        int nr = rows, nc = cols;
        if (d == 'D' || d == 'U') nr *= 2; else nc *= 2;
        if (d == 'D') {
            for (int r=0;r<rows;r++) for (int c=0;c<cols;c++) {
                nextPaper[rows-1-r][c] = flipVertical(paper[r][c]);
                nextPaper[rows+r][c] = paper[r][c];
            }
        } else if (d == 'U') {
            for (int r=0;r<rows;r++) for (int c=0;c<cols;c++) {
                nextPaper[r][c] = paper[r][c];
                nextPaper[2*rows-1-r][c] = flipVertical(paper[r][c]);
            }
        } else if (d == 'R') {
            for (int r=0;r<rows;r++) for (int c=0;c<cols;c++) {
                nextPaper[r][cols-1-c] = flipHorizontal(paper[r][c]);
                nextPaper[r][cols+c] = paper[r][c];
            }
        } else {
            for (int r=0;r<rows;r++) for (int c=0;c<cols;c++) {
                nextPaper[r][c] = paper[r][c];
                nextPaper[r][2*cols-1-c] = flipHorizontal(paper[r][c]);
            }
        }
        for (int r=0;r<nr;r++) for (int c=0;c<nc;c++) paper[r][c]=nextPaper[r][c];
        rows=nr; cols=nc;
    }
    for (int r=0;r<rows;r++) {
        for (int c=0;c<cols;c++) printf("%d%c", paper[r][c], c==cols-1?'\n':' ');
    }
    return 0;
}
