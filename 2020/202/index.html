6. 정답 C 코드

#include <stdio.h>

#define MAX_N 300000
#define MAX_EDGE 600000

typedef struct {
    int to;
    int next;
} Edge;

int head[MAX_N + 1];
Edge edges[MAX_EDGE];
int edgeCount;

int parent[MAX_N + 1];
int order[MAX_N];
int stack[MAX_N];
int subtreeSize[MAX_N + 1];

void addEdge(int from, int to) {
    edges[edgeCount].to = to;
    edges[edgeCount].next = head[from];

    head[from] = edgeCount;
    edgeCount++;
}

int main(void) {
    int N;

    scanf("%d", &N);

    for (int i = 1; i <= N; i++) {
        head[i] = -1;
    }

    edgeCount = 0;

    for (int i = 0; i < N - 1; i++) {
        int a;
        int b;

        scanf("%d %d", &a, &b);

        addEdge(a, b);
        addEdge(b, a);
    }

    int top = 0;
    int orderCount = 0;

    parent[1] = 0;
    stack[top++] = 1;

    while (top > 0) {
        int current = stack[--top];

        order[orderCount++] = current;

        for (
            int edgeIndex = head[current];
            edgeIndex != -1;
            edgeIndex = edges[edgeIndex].next
        ) {
            int next = edges[edgeIndex].to;

            if (next == parent[current]) {
                continue;
            }

            parent[next] = current;
            stack[top++] = next;
        }
    }

    for (int i = 1; i <= N; i++) {
        subtreeSize[i] = 1;
    }

    for (
        int index = orderCount - 1;
        index > 0;
        index--
    ) {
        int current = order[index];
        int p = parent[current];

        subtreeSize[p] += subtreeSize[current];
    }

    long long totalPairs =
        (long long)N * (N - 1) / 2;

    long long answer = 0;

    for (int current = 2; current <= N; current++) {
        long long s = subtreeSize[current];
        long long outside = N - s;

        long long unusedPairs =
            outside * (outside - 1) / 2;

        answer += totalPairs - unusedPairs;
    }

    printf("%lld\n", answer);

    return 0;
}
