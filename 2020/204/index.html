2단계. 길이 R 가능 여부 함수 작성

4단계. 같은 색깔의 중복 구간 제거

새 점의 고유 기여:
[max(y, previousY+R+1), min(y+R, nextY-1)]

구간 덧셈 + 전체 최댓값
segmentMaximum[1] == K 이면 성공

    mid = (low + high) / 2
    if (possible(mid)) high = mid
    else low = mid

7. 정답 C 코드

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_N 100000
#define MAX_K 100000
#define MAX_Y 500000
#define SEGMENT_SIZE (MAX_Y * 4 + 20)

typedef struct {
    int x;
    int y;
    int color;
    int id;
} Point;

int N;
int K;
Point points[MAX_N + 1];

int roots[MAX_K + 1];
int leftChild[MAX_N + 1];
int rightChild[MAX_N + 1];
unsigned int priority[MAX_N + 1];

int segmentMaximum[SEGMENT_SIZE];
int segmentLazy[SEGMENT_SIZE];

int comparePoint(
    const void *a,
    const void *b
) {
    const Point *p = (const Point *)a;
    const Point *q = (const Point *)b;

    if (p->x != q->x) {
        return p->x < q->x ? -1 : 1;
    }

    if (p->y != q->y) {
        return p->y < q->y ? -1 : 1;
    }

    if (p->id < q->id) {
        return -1;
    }

    if (p->id > q->id) {
        return 1;
    }

    return 0;
}

int keyLess(int a, int b) {
    if (points[a].y != points[b].y) {
        return points[a].y < points[b].y;
    }

    return a < b;
}

int mergeTreap(int a, int b) {
    if (a == 0) {
        return b;
    }

    if (b == 0) {
        return a;
    }

    if (priority[a] < priority[b]) {
        rightChild[a] =
            mergeTreap(
                rightChild[a],
                b
            );

        return a;
    }

    leftChild[b] =
        mergeTreap(
            a,
            leftChild[b]
        );

    return b;
}

int rotateRight(int root) {
    int newRoot = leftChild[root];

    leftChild[root] =
        rightChild[newRoot];

    rightChild[newRoot] = root;

    return newRoot;
}

int rotateLeft(int root) {
    int newRoot = rightChild[root];

    rightChild[root] =
        leftChild[newRoot];

    leftChild[newRoot] = root;

    return newRoot;
}

int insertTreap(
    int root,
    int node
) {
    if (root == 0) {
        return node;
    }

    if (keyLess(node, root)) {
        leftChild[root] =
            insertTreap(
                leftChild[root],
                node
            );

        if (
            priority[leftChild[root]]
            < priority[root]
        ) {
            root = rotateRight(root);
        }
    } else {
        rightChild[root] =
            insertTreap(
                rightChild[root],
                node
            );

        if (
            priority[rightChild[root]]
            < priority[root]
        ) {
            root = rotateLeft(root);
        }
    }

    return root;
}

int eraseTreap(
    int root,
    int node
) {
    if (root == node) {
        return mergeTreap(
            leftChild[root],
            rightChild[root]
        );
    }

    if (keyLess(node, root)) {
        leftChild[root] =
            eraseTreap(
                leftChild[root],
                node
            );
    } else {
        rightChild[root] =
            eraseTreap(
                rightChild[root],
                node
            );
    }

    return root;
}

int predecessor(
    int root,
    int node
) {
    int answer = 0;

    while (root != 0) {
        if (keyLess(root, node)) {
            answer = root;
            root = rightChild[root];
        } else {
            root = leftChild[root];
        }
    }

    return answer;
}

int successor(
    int root,
    int node
) {
    int answer = 0;

    while (root != 0) {
        if (keyLess(node, root)) {
            answer = root;
            root = leftChild[root];
        } else {
            root = rightChild[root];
        }
    }

    return answer;
}

void rangeAdd(
    int node,
    int start,
    int end,
    int left,
    int right,
    int value
) {
    if (
        right < start ||
        end < left
    ) {
        return;
    }

    if (
        left <= start &&
        end <= right
    ) {
        segmentMaximum[node] += value;
        segmentLazy[node] += value;
        return;
    }

    int middle = (start + end) / 2;

    rangeAdd(
        node * 2,
        start,
        middle,
        left,
        right,
        value
    );

    rangeAdd(
        node * 2 + 1,
        middle + 1,
        end,
        left,
        right,
        value
    );

    int childMaximum =
        segmentMaximum[node * 2]
        >
        segmentMaximum[node * 2 + 1]
        ?
        segmentMaximum[node * 2]
        :
        segmentMaximum[node * 2 + 1];

    segmentMaximum[node] =
        segmentLazy[node] +
        childMaximum;
}

void updateUniquePart(
    int pointIndex,
    int R,
    int delta
) {
    int color =
        points[pointIndex].color;

    int root = roots[color];

    int previous =
        predecessor(
            root,
            pointIndex
        );

    int next =
        successor(
            root,
            pointIndex
        );

    int left =
        points[pointIndex].y;

    int right =
        points[pointIndex].y + R;

    if (previous != 0) {
        int previousEnd =
            points[previous].y + R;

        if (previousEnd + 1 > left) {
            left = previousEnd + 1;
        }
    }

    if (next != 0) {
        int beforeNext =
            points[next].y - 1;

        if (beforeNext < right) {
            right = beforeNext;
        }
    }

    if (left <= right) {
        rangeAdd(
            1,
            1,
            MAX_Y,
            left,
            right,
            delta
        );
    }
}

void addPoint(
    int index,
    int R
) {
    int color =
        points[index].color;

    leftChild[index] = 0;
    rightChild[index] = 0;

    updateUniquePart(
        index,
        R,
        1
    );

    roots[color] =
        insertTreap(
            roots[color],
            index
        );
}

void removePoint(
    int index,
    int R
) {
    int color =
        points[index].color;

    updateUniquePart(
        index,
        R,
        -1
    );

    roots[color] =
        eraseTreap(
            roots[color],
            index
        );
}

int possible(int R) {
    memset(
        roots,
        0,
        sizeof(int) * (K + 1)
    );

    memset(
        segmentMaximum,
        0,
        sizeof(segmentMaximum)
    );

    memset(
        segmentLazy,
        0,
        sizeof(segmentLazy)
    );

    int left = 1;
    int right = 1;

    while (left <= N) {
        int xLeft =
            points[left].x;

        while (
            right <= N &&
            points[right].x <= xLeft + R
        ) {
            addPoint(right, R);
            right++;
        }

        if (
            segmentMaximum[1] == K
        ) {
            return 1;
        }

        while (
            left <= N &&
            points[left].x == xLeft
        ) {
            removePoint(left, R);
            left++;
        }
    }

    return 0;
}

unsigned int randomState =
    2463534242u;

unsigned int xorshift32(void) {
    randomState ^= randomState << 13;
    randomState ^= randomState >> 17;
    randomState ^= randomState << 5;

    return randomState;
}

int main(void) {
    scanf("%d %d", &N, &K);

    for (int i = 1; i <= N; i++) {
        scanf(
            "%d %d %d",
            &points[i].x,
            &points[i].y,
            &points[i].color
        );

        points[i].id = i;
    }

    qsort(
        points + 1,
        N,
        sizeof(Point),
        comparePoint
    );

    for (int i = 1; i <= N; i++) {
        priority[i] =
            xorshift32();
    }

    int low = -1;
    int high = 250000;

    while (high - low > 1) {
        int middle =
            low +
            (high - low) / 2;

        if (possible(middle)) {
            high = middle;
        } else {
            low = middle;
        }
    }

    printf("%d\n", high);

    return 0;
}
