정올 11761 승리 스왑

s1을 t로 만드는 과정을 한 단계씩 확인하는 인터랙티브 강의자료

1. 문제 핵심

목표는 여러 문자열 중 s1을 목표 문자열 t와 같게 만드는 것입니다.

1번 연산: 1 x p q x번 문자열 안에서 p번째 문자와 q번째 문자를 교환 2번 연산: 2 x y k x번 문자열과 y번 문자열의 k번째 문자를 교환
왼쪽부터 오른쪽으로 s1의 문자를 하나씩 t와 같게 만들면 됩니다.

2. 핵심 전략

i번째 위치를 맞춘다고 하자. 필요한 문자 = t[i] 1. s1의 i보다 오른쪽에서 필요한 문자를 찾는다. 찾으면 1번 연산 한 번으로 해결한다. 2. s1 뒤쪽에 없으면 다른 문자열에서 필요한 문자를 찾는다. 다른 문자열 안에서 i번째 위치로 옮긴 뒤, s1과 같은 위치끼리 교환한다.
이미 맞춘 앞부분은 건드리면 안 됩니다. 그래서 s1 안에서는 항상 현재 위치보다 오른쪽만 찾습니다.

5. 최종 정답 C 코드

#include 
#include 

#define MAXN 1000
#define MAXM 1000
#define MAXOP 2005

typedef struct {
    int type;
    int a;
    int b;
    int c;
} Operation;

int N, M;
char t[MAXM + 5];
char s[MAXN + 5][MAXM + 5];
Operation ops[MAXOP];
int opCount;

void swap_char(char *a, char *b) {
    char temp = *a;
    *a = *b;
    *b = temp;
}

void add_operation(int type, int a, int b, int c) {
    ops[opCount].type = type;
    ops[opCount].a = a;
    ops[opCount].b = b;
    ops[opCount].c = c;
    opCount++;
}

int main() {
    int T;
    scanf("%d", &T);

    while (T--) {
        scanf("%d %d", &N, &M);
        scanf("%s", t);

        for (int i = 0; i < N; i++) scanf("%s", s[i]);

        opCount = 0;

        for (int i = 0; i < M; i++) {
            if (s[0][i] == t[i]) continue;

            char need = t[i];
            int found = 0;

            for (int j = i + 1; j < M; j++) {
                if (s[0][j] == need) {
                    add_operation(1, 1, i + 1, j + 1);
                    swap_char(&s[0][i], &s[0][j]);
                    found = 1;
                    break;
                }
            }

            if (found) continue;

            int row = -1, col = -1;

            for (int x = 1; x < N; x++) {
                for (int j = 0; j < M; j++) {
                    if (s[x][j] == need) {
                        row = x;
                        col = j;
                        found = 1;
                        break;
                    }
                }
                if (found) break;
            }

            if (col != i) {
                add_operation(1, row + 1, col + 1, i + 1);
                swap_char(&s[row][col], &s[row][i]);
            }

            add_operation(2, 1, row + 1, i + 1);
            swap_char(&s[0][i], &s[row][i]);
        }

        printf("%d\n", opCount);

        for (int i = 0; i < opCount; i++) {
            printf("%d %d %d %d\n", ops[i].type, ops[i].a, ops[i].b, ops[i].c);
        }
    }

    return 0;
}

6. 연산 수가 2M 이하인 이유

각 위치마다 이미 맞음: 0번 s1 뒤쪽에서 찾음: 1번 다른 문자열 같은 위치에서 찾음: 1번 다른 문자열 다른 위치에서 찾음: 2번 따라서 전체 연산 수 ≤ 2M
