다익스트라 알고리즘이란?

세영코딩연구소 · 정올 골드 | 음수 없는 가중치 그래프의 단일 출발 최단 경로

핵심 아이디어
"아직 확정되지 않은 노드 중 dist가 가장 작은 것을 꺼내서 이웃을 완화(Relaxation)"
탐욕(Greedy) 알고리즘 — 한 번 확정된 노드는 다시 갱신되지 않아!

동작 3단계

① 초기화
출발 노드 dist=0, 나머지 dist=INF
출발 노드를 우선순위 큐에 push
② 큐에서 최소값 꺼내기
dist[v] 보다 큰 구식 값이면 skip
해당 노드를 "확정(visited)"
③ 이웃 완화(Relaxation)
dist[v]+w < dist[u] 이면 갱신
갱신된 노드를 큐에 push

복잡도 비교

구현시간복잡도공간
배열 탐색 (brute)O(V²)O(V)
이진 힙 + 인접리스트O((V+E)logV)O(V+E)
피보나치 힙O(E + VlogV)O(V+E)
정올에서는 이진 힙 + 인접리스트가 표준!
V, E ≤ 100,000일 때 충분히 빠름.

왜 음수 간선은 안 되나?

다익스트라는 "한 번 확정된 노드의 거리는 최단"이라고 가정해. 음수 간선이 있으면 이미 확정된 노드를 더 짧은 경로로 재방문할 수 있어서 이 가정이 깨져!

음수 간선 → 벨만-포드(Bellman-Ford) O(VE) 사용
음수 사이클 탐지 → 벨만-포드 V번째 반복에서 갱신 확인

다익스트라 vs 다른 최단경로

/* ── 간선 추가 (정적 연결 리스트 패턴) ── */
void addEdge(int u, int v, int w) {
    to[ecnt]  = v;
    wt[ecnt]  = w;
    nxt[ecnt] = head[u]; // 기존 head를 next로
    head[u]   = ecnt++;  // 새 간선이 head
}
/* head[u]에서 nxt[]를 따라가면 u의 모든 이웃 순회 */
/* 역방향도 필요하면 addEdge(v, u, w)도 호출 */

/* ── 초기화 ── */
memset(head, -1, sizeof(head)); // -1 = 간선 없음
memset(dist, 0x3f, sizeof(dist)); // INF로 초기화
memset(visited, 0, sizeof(visited));

/* ── 순회 패턴 ── */
for (int e = head[v]; e != -1; e = nxt[e]) {
    int u = to[e], w = wt[e];
    // u는 v의 이웃, 가중치 w
}
c_dijkstra_03_minheap.c
/* ── Min-Heap 구현 ── */
void heapUp(int i) {
    while (i > 0) {
        int p = (i-1)/2;
        if (heap[p].dist <= heap[i].dist) break;
        HeapNode tmp = heap[p]; heap[p]=heap[i]; heap[i]=tmp;
        i = p;
    }
}
void heapDown(int i) {
    while (2*i+1 < heapSz) {
        int l=2*i+1, r=2*i+2, s=i;
        if(l<heapSz && heap[l].dist<heap[s].dist) s=l;
        if(r<heapSz && heap[r].dist<heap[s].dist) s=r;
        if(s==i) break;
        HeapNode tmp=heap[i]; heap[i]=heap[s]; heap[s]=tmp;
        i=s;
    }
}
void push(int d, int v){heap[heapSz++]=(HeapNode){d,v};heapUp(heapSz-1);}
HeapNode pop(){HeapNode r=heap[0];heap[0]=heap[--heapSz];heapDown(0);return r;}
c_dijkstra_04_main.c
void dijkstra(int src) {
    dist[src] = 0;
    push(0, src);
    while (heapSz > 0) {
        HeapNode cur = pop();
        int v = cur.node, d = cur.dist;
        // 구식 값 무시!
        if (visited[v]) continue;
        visited[v] = 1;
        for (int e=head[v]; e!=-1; e=nxt[e]) {
            int u=to[e], nd=d+wt[e];
            if (nd < dist[u]) {
                dist[u]=nd;
                push(nd, u);
            }
        }
    }
}

cpp_dijkstra_basic.cpp
#include <bits/stdc++.h>
using namespace std;
const int INF = 0x3f3f3f3f;
using pii = pair<int,int>;

// 인접 리스트 — vector로 간단!
vector<pii> adj[100001]; // {가중치, 목적지}

void dijkstra(int src, int n, vector<int>& dist) {
    dist.assign(n+1, INF);
    dist[src] = 0;

    // Min-Heap: {거리, 노드}
    priority_queue<pii, vector<pii>, greater<pii>> pq;
    pq.push({0, src});

    while (!pq.empty()) {
        auto [d, v] = pq.top(); pq.pop();
        if (d > dist[v]) continue; // 구식값 무시!
        for (auto [w, u] : adj[v]) {
            if (dist[v]+w < dist[u]) {
                dist[u] = dist[v]+w;
                pq.push({dist[u], u});
            }
        }
    }
}
cpp_dijkstra_compact.cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using pli = pair<ll,int>;
const ll INF = 1e18;

int main() {
    ios_base::sync_with_stdio(false); cin.tie(NULL);
    int n, m, s; cin >> n >> m >> s;

    vector<vector<pair<int,int>>> g(n+1);
    for (int i=0;i<=m;i++){int u,v,w;cin>>u>>v>>w;g[u].push_back({w,v});}

    vector<ll> d(n+1, INF); d[s]=0;
    priority_queue<pli,vector<pli>,greater<>> pq; pq.push({0,s});

    while(!pq.empty()) {
        auto [dist,v]=pq.top(); pq.pop();
        if(dist>d[v]) continue;
        for(auto[w,u]:g[v]) if(d[v]+w<d[u]){d[u]=d[v]+w;pq.push({d[u],u});}
    }
    for(int i=1;i<=n;i++) cout<<(d[i]==INF?"INF":to_string(d[i]))<<"\n";
}

격자(Grid) 다익스트라

2D 격자에서 가중치가 있는 최단 경로!

격자에서 각 칸에 이동 비용이 있을 때 Dijkstra 적용.
BFS는 모든 비용이 같을 때, Dijkstra는 칸마다 비용이 다를 때 사용해!
C 격자 Dijkstra
#define INF 0x3f3f3f3f
int cost[N][N], dist[N][N];
typedef struct{int d,r,c;}Node;
Node heap[N*N]; int hsz=0;
/* Min-Heap push/pop 생략 */

int dr[]={-1,1,0,0};
int dc[]={ 0,0,-1,1};

memset(dist,0x3f,sizeof(dist));
dist[0][0]=cost[0][0];
push(cost[0][0],0,0);

while(hsz){
  Node c=pop();
  if(c.d>dist[c.r][c.c]) continue;
  for(int d=0;d<4;d++){
    int nr=c.r+dr[d];
    int nc=c.c+dc[d];
    if(nr<0||nr>=N) continue;
    if(nc<0||nc>=N) continue;
    int nd=c.d+cost[nr][nc];
    if(nd<dist[nr][nc]){
      dist[nr][nc]=nd;
      push(nd,nr,nc);
    }
  }
}
C++ 격자 Dijkstra
using T3 = tuple<int,int,int>; // {d,r,c}
priority_queue<T3,vector<T3>,
  greater<>> pq;

int dr[]={-1,1,0,0};
int dc[]={ 0,0,-1,1};

vector<vector<int>> d(N,vector<int>(N,INF));
d[0][0]=cost[0][0];
pq.push({d[0][0],0,0});

while(!pq.empty()){
  auto[dist,r,c]=pq.top();pq.pop();
  if(dist>d[r][c]) continue;
  for(int k=0;k<4;k++){
    int nr=r+dr[k], nc=c+dc[k];
    if(nr<0||nr>=N||nc<0||nc>=N)
      continue;
    int nd=dist+cost[nr][nc];
    if(nd<d[nr][nc]){
      d[nr][nc]=nd;
      pq.push({nd,nr,nc});
    }
  }
}
tuple vs pair 선택
노드 정보가 2개 → pair<int,int> (dist, node)
격자처럼 3개 → tuple<int,int,int> (dist, row, col)
C++17의 구조적 바인딩 auto [d,r,c] = pq.top()으로 깔끔하게!

C vs C++ 성능 분석

코드 길이, 속도, 메모리, 안전성 — 실전 비교!

C 언어 장점

  • 메모리 레이아웃 직접 제어
  • 캐시 친화적인 배열 기반 힙
  • 동적 할당 없음 → 메모리 예측 가능
  • 컴파일러 최적화가 더 적극적
  • 포인터 조작으로 극한 최적화 가능

C++ STL 장점

  • 코드 길이 1/3 수준으로 단축
  • STL은 검증된 버그 없는 구현
  • 힙 직접 구현 실수 제거
  • C++17 구조적 바인딩으로 가독성↑
  • 정올 시간 초과 기준 내 충분히 빠름

정올 실전 벤치마크 (V=100K, E=300K)
